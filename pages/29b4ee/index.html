# 描述

部署工具crylogger。

（前言：配置过程还挺折磨，折腾了两三天，包括在主机尝试配置，在docker中运行等等。感觉配置过程太依赖AI了，有些错误解决方法其实网络上面的经验帖更准确一些。）

（在部署中遇到了许多问题，尝试了许多方案。下面写的是成功的，顺便会在最后附上猜测可能会成功的方案）

# 步骤（还需要再验证一下）

接下来进行的操作实际较长，为了防止本地与远程服务器断联风险，可以使用tmux命令。接下来的命令都在tmux窗口内进行。

tmux new -s <name> 	# 创建新窗口，例子 tmux new -s crylogger
tmux attach -t <name> 	# 进入窗口，例子 tmux attach -t crylogger

# 1、配置环境

  • 编译aosp。

首先需下载ubuntu18.04docker，需要提前配置好docker国内镜像源。

编写docker file如下

FROM ubuntu:18.04

# 避免安装软件时弹出交互式配置界面卡住
ENV DEBIAN_FRONTEND=noninteractive

# 1. 将软件源替换为阿里云镜像
RUN sed -i 's/archive.ubuntu.com/mirrors.aliyun.com/g' /etc/apt/sources.list && \
    sed -i 's/security.ubuntu.com/mirrors.aliyun.com/g' /etc/apt/sources.list

# 2. 安装 AOSP 编译所需的核心依赖包（增加了 locales）
RUN apt-get update && apt-get install -y \
    git-core gnupg flex bison build-essential \
    zip curl zlib1g-dev gcc-multilib g++-multilib \
    libc6-dev-i386 lib32ncurses5-dev x11proto-core-dev \
    libx11-dev lib32z1-dev libgl1-mesa-dev libxml2-utils \
    xsltproc unzip fontconfig python2.7 python-pip \
    ccache rsync sudo locales wget openjdk-8-jdk vim lzip python3-pip python3 \
    libpulse0 libgl1-mesa-glx libgl1-mesa-dri libx11-6 libxcb1 libxcb-shm0 libxcb-xfixes0 \
    && rm -rf /var/lib/apt/lists/*

# 配置 sudo 组免密（关键）
RUN echo '%sudo ALL=(ALL) NOPASSWD:ALL' >> /etc/sudoers

# 3. 生成 en_US.UTF-8 locale 数据（彻底解决 warning）
RUN locale-gen en_US.UTF-8

# 4. 将 python2 设为默认
RUN ln -sf /usr/bin/python2.7 /usr/bin/python

# 5. 安装 repo 工具
RUN curl -o /usr/local/bin/repo https://mirrors.tuna.tsinghua.edu.cn/git/git-repo && \
    chmod a+x /usr/local/bin/repo

# 6. 设置 UTF-8 环境
ENV LANG=en_US.UTF-8 \
    LANGUAGE=en_US:en \
    LC_ALL=en_US.UTF-8

# 7. 配置 ccache 环境变量
ENV USE_CCACHE=1 \
    CCACHE_DIR=/ccache \
    CCACHE_EXEC=/usr/bin/ccache

# 创建普通用户并切换（放在最后）
ARG UID=1000
ARG GID=1000
ARG USERNAME=builduser
RUN groupadd -g ${GID} ${USERNAME} 2>/dev/null || true && \
    useradd -m -u ${UID} -g ${GID} -G sudo -s /bin/bash ${USERNAME}
# 如果需要创建 /aosp 并改属主，可以在这里做（root 时创建并 chown）
RUN mkdir -p /aosp && chown ${USERNAME}:${USERNAME} /aosp
USER ${USERNAME}
WORKDIR /aosp

接下来修改 android-emu/art/runtime/mem_map.cc 文件，防止地址随机化导致编译不成功。

# 假设当前在crylogger目录下面
vim android-emu/art/runtime/mem_map.cc
# 在vim编辑器中，输入 
:set nu 		# 设置行号显示
507G		#跳转到文件的第507行

# 在 flags |= MAP_FIXED; 这行后面添加以下代码
#if !defined(ART_TARGET)
  } else if (expected_ptr) {
#define MAP_FIXED_NOREPLACE 0x100000
    flags |= MAP_FIXED_NOREPLACE;
#endif

即原来的上下文是
  flags |= MAP_FIXED;
  } else {
    CHECK_EQ(0, flags & MAP_FIXED);
  
修改后变成了
    flags |= MAP_FIXED;
#if !defined(ART_TARGET)
  } else if (expected_ptr) {
#define MAP_FIXED_NOREPLACE 0x100000
    flags |= MAP_FIXED_NOREPLACE;
#endif
  } else {
    CHECK_EQ(0, flags & MAP_FIXED);

给当前用户配置kvm组，防止之后启动模拟器不成功

# 输入以下命令，查看kvm的权限
ls -l /dev/kvm
# 输出可能类似以下所示
crw-rw---- 1 root 994 10, 232 Jul  1 06:29 /dev/kvm
# 手动创建组并添加用户
sudo groupadd -g 994 kvm
sudo usermod -aG kvm $(whoami)
newgrp kvm   # 刷新组

# 如果输出类似 crw-rw---- 1 root kvm 10, 232 ... ，那么不需要创建组，执行下面命令即可
sudo usermod -aG kvm $USER
newgrp kvm    # 刷新组
